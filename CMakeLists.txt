cmake_minimum_required(VERSION 3.20)
project(Group2Engine VERSION 0.1.0 LANGUAGES CXX)

# Provisional baseline: review with the team before using C++26 features.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)

# ENet v1.3.18 (MIT): networking transport, see docs/DECISIONS.md ADR NET-2.
include(FetchContent)
FetchContent_Declare(enet GIT_REPOSITORY https://github.com/lsalzman/enet.git GIT_TAG v1.3.18 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(enet)
# ENet's CMake only sets a directory include path and links Winsock for MinGW; fix both for consumers.
target_include_directories(enet SYSTEM INTERFACE ${enet_SOURCE_DIR}/include)
if(WIN32)
    target_link_libraries(enet INTERFACE ws2_32 winmm)
endif()

function(group2_target target)
    target_compile_features(${target} PUBLIC cxx_std_20)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_library(engine_core INTERFACE)
target_include_directories(engine_core INTERFACE engine/include)
target_compile_features(engine_core INTERFACE cxx_std_20)
foreach(module IN ITEMS architecture rendering physics networking ai tools interaction infrastructure)
    add_library(engine_${module} STATIC engine/src/${module}.cpp)
    target_link_libraries(engine_${module} PUBLIC engine_core)
    group2_target(engine_${module})
endforeach()
# PhysicsSystem reads Body/Collider components from the World.
target_link_libraries(engine_physics PUBLIC engine_architecture)

target_sources(engine_tools PRIVATE engine/src/tiled_importer.cpp)
target_include_directories(engine_tools SYSTEM PRIVATE third_party/nlohmann_json/include)

target_sources(engine_infrastructure PRIVATE engine/src/resource_files.cpp engine/src/configuration.cpp)
target_include_directories(engine_infrastructure SYSTEM PRIVATE third_party/nlohmann_json/include)
target_sources(engine_networking PRIVATE engine/src/enet_session.cpp)
target_link_libraries(engine_networking PRIVATE enet)

add_library(elemental_coop_game STATIC games/elemental_coop/src/game.cpp)
target_include_directories(elemental_coop_game PUBLIC games/elemental_coop/include)
target_link_libraries(elemental_coop_game PUBLIC engine_architecture engine_physics engine_interaction)
group2_target(elemental_coop_game)

add_library(elemental_coop_net STATIC games/elemental_coop/src/net.cpp)
target_link_libraries(elemental_coop_net PUBLIC elemental_coop_game engine_networking)
group2_target(elemental_coop_net)

add_executable(engine_demo apps/headless_demo.cpp)
target_link_libraries(engine_demo PRIVATE elemental_coop_net elemental_coop_game engine_rendering engine_networking engine_ai engine_tools engine_infrastructure)
group2_target(engine_demo)

add_executable(infrastructure_demo apps/infrastructure_demo.cpp)
target_link_libraries(infrastructure_demo PRIVATE engine_infrastructure)
group2_target(infrastructure_demo)

add_executable(ai_demo apps/ai_demo.cpp)
target_link_libraries(ai_demo PRIVATE engine_ai)
group2_target(ai_demo)

if(BUILD_TESTING)
    configure_file(tests/fixtures/tiled_level.json fixtures/tiled_level.json COPYONLY)
    foreach(suite IN ITEMS architecture rendering physics networking ai tools interaction infrastructure resource_files configuration gameplay integration net_session coop_net)
        add_executable(test_${suite} tests/${suite}_test.cpp)
        target_include_directories(test_${suite} PRIVATE tests)
        target_link_libraries(test_${suite} PRIVATE elemental_coop_net elemental_coop_game engine_rendering engine_networking engine_ai engine_tools engine_infrastructure)
        group2_target(test_${suite})
        add_test(NAME ${suite} COMMAND test_${suite})
    endforeach()
endif()

# Optional backend: the default headless build remains dependency-free.
option(GROUP2_BUILD_SDL_RENDERER "Build SDL3 rendering backend and demo" OFF)
option(GROUP2_FETCH_SDL "Download the project's pinned SDL3 version" OFF)
if(GROUP2_BUILD_SDL_RENDERER)
    if(NOT TARGET SDL3::SDL3)
        if(GROUP2_FETCH_SDL)
            if(POLICY CMP0135)
                cmake_policy(SET CMP0135 NEW)
            endif()
            include(FetchContent)
            set(SDL_SHARED OFF CACHE BOOL "" FORCE)
            set(SDL_STATIC ON CACHE BOOL "" FORCE)
            set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
            FetchContent_Declare(SDL3
                URL https://www.libsdl.org/release/SDL3-3.2.28.tar.gz
                URL_HASH SHA256=1330671214d146f8aeb1ed399fc3e081873cdb38b5189d1f8bb6ab15bbc04211)
            FetchContent_MakeAvailable(SDL3)
        else()
            find_package(SDL3 3.2 CONFIG REQUIRED)
        endif()
    endif()
    add_library(engine_sdl_rendering STATIC engine/src/sdl_renderer.cpp)
    target_link_libraries(engine_sdl_rendering PUBLIC engine_rendering SDL3::SDL3 PRIVATE engine_infrastructure)
    group2_target(engine_sdl_rendering)
    add_executable(rendering_demo apps/rendering_demo.cpp)
    target_link_libraries(rendering_demo PRIVATE engine_sdl_rendering)
    group2_target(rendering_demo)
    add_custom_command(TARGET rendering_demo POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/assets/rendering" "$<TARGET_FILE_DIR:rendering_demo>/rendering-assets"
        VERBATIM)
    if(BUILD_TESTING)
        add_executable(test_sdl_rendering tests/sdl_rendering_test.cpp)
        target_include_directories(test_sdl_rendering PRIVATE tests)
        target_link_libraries(test_sdl_rendering PRIVATE engine_sdl_rendering)
        group2_target(test_sdl_rendering)
        add_test(NAME sdl_rendering COMMAND test_sdl_rendering)
        set_tests_properties(sdl_rendering PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
        add_test(NAME rendering_demo_smoke COMMAND rendering_demo --smoke-test)
        set_tests_properties(rendering_demo_smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy")
    endif()
    if(NOT GROUP2_UI_STANDALONE)
        add_subdirectory(apps/ui_prototype)
    endif()
endif()
